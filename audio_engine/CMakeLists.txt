cmake_minimum_required(VERSION 3.20)
project(audio_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(AUDIO_ENGINE_SOURCES
    src/AudioEngine.cpp src/DacDeviceCatalog.cpp src/DsfParser.cpp src/IphoneDsdDecoder.cpp
    src/library/MusicLibrary.cpp src/library/LibraryMetadata.cpp src/library/LibraryDatabase.cpp
    third_party/sqlite/sqlite3.c
    src/decoders/FlacDecoder.cpp src/decoders/WavDecoder.cpp src/decoders/Mp3Decoder.cpp)
if(ANDROID)
    list(APPEND AUDIO_ENGINE_SOURCES src/sinks/AAudioSink.cpp)
elseif(APPLE)
    list(APPEND AUDIO_ENGINE_SOURCES src/decoders/AppleAudioDecoder.cpp
        src/decoders/CoreAudioPcmDecoder.cpp src/sinks/AudioQueueSink.cpp)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        list(APPEND AUDIO_ENGINE_SOURCES src/sinks/RemoteIOSink.cpp)
    endif()
else()
    find_package(SDL2 REQUIRED)
    list(APPEND AUDIO_ENGINE_SOURCES src/sinks/SdlAudioSink.cpp)
    if(UNIX)
        list(APPEND AUDIO_ENGINE_SOURCES src/sinks/AlsaAudioSink.cpp)
        find_package(PkgConfig QUIET)
        if(PKG_CONFIG_FOUND)
            pkg_check_modules(ALSA QUIET alsa)
        endif()
    endif()
endif()

set_source_files_properties(third_party/sqlite/sqlite3.c PROPERTIES COMPILE_DEFINITIONS "SQLITE_THREADSAFE=1;SQLITE_OMIT_LOAD_EXTENSION=1;SQLITE_DEFAULT_FOREIGN_KEYS=0")
if(NOT WIN32)
    set_property(SOURCE third_party/sqlite/sqlite3.c APPEND PROPERTY COMPILE_DEFINITIONS "SQLITE_API=__attribute__((visibility(\"hidden\")))")
endif()

# Object target also supplies a self-contained iOS C ABI archive.
add_library(audio_engine_objects OBJECT ${AUDIO_ENGINE_SOURCES})
add_library(audio_engine STATIC $<TARGET_OBJECTS:audio_engine_objects>)
foreach(engine_target IN ITEMS audio_engine_objects audio_engine)
    target_compile_features(${engine_target} PUBLIC cxx_std_20)
    target_include_directories(${engine_target} PUBLIC include PRIVATE third_party ${SDL2_INCLUDE_DIRS})
    if(ANDROID)
        target_link_libraries(${engine_target} PUBLIC aaudio log)
    elseif(APPLE)
        target_link_libraries(${engine_target} PUBLIC "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
    else()
        target_link_libraries(${engine_target} PUBLIC ${SDL2_LIBRARIES})
        if(UNIX)
            target_link_libraries(${engine_target} PUBLIC pthread)
            if(ALSA_FOUND)
                target_include_directories(${engine_target} PRIVATE ${ALSA_INCLUDE_DIRS})
                target_link_libraries(${engine_target} PUBLIC ${ALSA_LIBRARIES})
            endif()
        endif()
    endif()
endforeach()
