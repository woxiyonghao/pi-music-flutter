cmake_minimum_required(VERSION 3.20)
project(hifi_audio_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
add_subdirectory(audio_engine)

option(HIFI_BUILD_FFI "Build the public C ABI" ON)
if(HIFI_BUILD_FFI)
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_library(hifi_audio_ffi STATIC src/hifi_audio_ffi.cpp $<TARGET_OBJECTS:audio_engine_objects>)
    else()
        add_library(hifi_audio_ffi SHARED src/hifi_audio_ffi.cpp)
    endif()
    target_include_directories(hifi_audio_ffi PUBLIC include)
    target_link_libraries(hifi_audio_ffi PRIVATE audio_engine)
    set_target_properties(hifi_audio_ffi PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
    if(APPLE)
        set_target_properties(hifi_audio_ffi PROPERTIES MACOSX_RPATH ON INSTALL_NAME_DIR "@rpath")
    endif()
    if(WIN32 AND TARGET SDL2::SDL2)
        get_target_property(HIFI_SDL_TYPE SDL2::SDL2 TYPE)
        if(HIFI_SDL_TYPE STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET hifi_audio_ffi POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:hifi_audio_ffi>)
        endif()
    endif()
    install(TARGETS hifi_audio_ffi ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
    install(FILES include/hifi_audio_ffi.h DESTINATION include)
endif()

if(BUILD_TESTING)
    add_executable(music_library_test audio_engine/tests/music_library_test.cpp)
    target_link_libraries(music_library_test PRIVATE hifi_audio_ffi audio_engine)
    add_test(NAME music_library COMMAND music_library_test)
    add_executable(pcm_rate_decoder_test audio_engine/tests/pcm_rate_decoder_test.cpp)
    target_link_libraries(pcm_rate_decoder_test PRIVATE audio_engine)
    add_test(NAME pcm_rate_decoder COMMAND pcm_rate_decoder_test)
    add_executable(iphone_dsd_decoder_test audio_engine/tests/iphone_dsd_decoder_test.cpp)
    target_link_libraries(iphone_dsd_decoder_test PRIVATE audio_engine)
    add_test(NAME iphone_dsd_decoder COMMAND iphone_dsd_decoder_test)
    add_executable(worker_thread_test audio_engine/tests/worker_thread_test.cpp)
    target_link_libraries(worker_thread_test PRIVATE audio_engine)
    add_test(NAME worker_thread COMMAND worker_thread_test)
    if(HIFI_BUILD_FFI)
        add_executable(platform_contract_test audio_engine/tests/platform_contract_test.cpp)
        target_link_libraries(platform_contract_test PRIVATE hifi_audio_ffi audio_engine)
        add_test(NAME platform_contract COMMAND platform_contract_test)
    endif()
    add_executable(graphic_equalizer_test audio_engine/tests/graphic_equalizer_test.cpp)
    target_link_libraries(graphic_equalizer_test PRIVATE audio_engine)
    add_test(NAME graphic_equalizer COMMAND graphic_equalizer_test)
    if(APPLE AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_executable(core_audio_playback_test audio_engine/tests/core_audio_playback_test.cpp)
        target_link_libraries(core_audio_playback_test PRIVATE audio_engine)
        add_test(NAME core_audio_pcm COMMAND core_audio_playback_test)
    endif()
endif()
