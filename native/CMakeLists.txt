cmake_minimum_required(VERSION 3.20)
project(hifi_platform_audio LANGUAGES C CXX)
# Integration only. All audio implementation lives on dev/audio_engine.
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(NOT HIFI_DART_EXECUTABLE)
    find_program(HIFI_DART_LAUNCHER NAMES dart dart.bat REQUIRED)
    get_filename_component(HIFI_DART_BIN "${HIFI_DART_LAUNCHER}" DIRECTORY)
    if(WIN32)
        set(HIFI_DART_EXECUTABLE "${HIFI_DART_BIN}/cache/dart-sdk/bin/dart.exe")
    else()
        set(HIFI_DART_EXECUTABLE "${HIFI_DART_BIN}/cache/dart-sdk/bin/dart")
    endif()
    if(NOT EXISTS "${HIFI_DART_EXECUTABLE}")
        set(HIFI_DART_EXECUTABLE "${HIFI_DART_LAUNCHER}")
    endif()
endif()
execute_process(COMMAND "${HIFI_DART_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/fetch_audio_engine.dart"
    RESULT_VARIABLE HIFI_FETCH_STATUS OUTPUT_VARIABLE HIFI_ENGINE_SOURCE OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT HIFI_FETCH_STATUS EQUAL 0)
    message(FATAL_ERROR "Could not fetch the locked audio_engine SDK")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../audio_engine.lock.json")
set(BUILD_TESTING OFF CACHE BOOL "Engine tests run on dev/audio_engine" FORCE)
set(HIFI_BUILD_FFI ON CACHE BOOL "Build C ABI" FORCE)
add_subdirectory("${HIFI_ENGINE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/engine")
set_target_properties(hifi_audio_ffi PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
